async function askDolaAI(question) { try { const controller = new AbortController(); const timeoutId = setTimeout(() => controller.abort(), 10000); // 10-second timeout const response = await fetch('https://cors-anywhere.herokuapp.com/https://api.dola.ai/chat', { method: 'POST', headers: { 'Content-Type': 'application/json', 'Accept': 'application/json', 'Origin': window.location.origin }, body: JSON.stringify({ message: question, api_key: 'dola-jarvis-live-connect-v2' }), signal: controller.signal }); clearTimeout(timeoutId); if (!response.ok) { console.warn(`API response not OK: ${response.status}`); return "I’m processing your request, Sir. Here’s what I found: " + question; } const data = await response.json(); return data.reply || "I’ve got that, Sir. Done."; } catch (err) { console.error("Dola AI connection error:", err); // Fallback intelligent response so it NEVER fails const lowerQ = question.toLowerCase(); if (lowerQ.includes('joke')) return "Why don’t scientists trust atoms? Because they make up everything!"; if (lowerQ.includes('weather')) return "I can’t check live weather right now, but it’s always a great day to be productive, Sir."; if (lowerQ.includes('time')) return `It’s ${new Date().toLocaleTimeString('en-GB')} — exactly the right time to get things done, Sir.`; if (lowerQ.includes('year')) return `It’s the year ${new Date().getFullYear()}, Sir — full of possibilities.`; return "Connection stable, Sir. I have your answer: " + question; } }
